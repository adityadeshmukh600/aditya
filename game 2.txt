using UnityEngine;
using UnityEngine.UI;

public class BattleRoyaleGame : MonoBehaviour
{
    // Player settings
    public float moveSpeed = 5f;
    public float jumpForce = 5f;
    public GameObject[] weapons;  // Array of weapon prefabs
    public Transform shootPoint;
    public float bulletSpeed = 10f;
    public int currentWeaponIndex = 0;
    public float health = 100f;
    public Text healthText;

    // Safe Zone settings
    public Transform safeZone;
    public float shrinkSpeed = 0.5f;
    public float minSize = 10f;

    // Graphics settings
    public Dropdown qualityDropdown;
    public Dropdown resolutionDropdown;

    private float horizontal, vertical;
    private bool isGrounded;
    private Rigidbody rb;

    void Start()
    {
        rb = GetComponent<Rigidbody>();
        qualityDropdown.onValueChanged.AddListener(OnQualityChanged);
        resolutionDropdown.onValueChanged.AddListener(OnResolutionChanged);
        qualityDropdown.value = QualitySettings.GetQualityLevel();
        resolutionDropdown.value = GetCurrentResolutionIndex();
    }

    void Update()
    {
        // Player movement
        horizontal = Input.GetAxis("Horizontal");
        vertical = Input.GetAxis("Vertical");
        Vector3 movement = new Vector3(horizontal, 0f, vertical) * moveSpeed * Time.deltaTime;
        transform.Translate(movement, Space.World);

        // Jumping
        if (isGrounded && Input.GetKeyDown(KeyCode.Space))
        {
            rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
        }

        // Weapon switching
        if (Input.GetKeyDown(KeyCode.Q)) SwitchWeapon(-1);
        if (Input.GetKeyDown(KeyCode.E)) SwitchWeapon(1);

        // Shooting
        if (Input.GetMouseButtonDown(0)) Shoot();

        // Safe zone shrinking
        if (safeZone.localScale.x > minSize)
        {
            safeZone.localScale -= new Vector3(shrinkSpeed, 0, shrinkSpeed) * Time.deltaTime;
        }

        // Health management
        if (IsOutsideSafeZone()) health -= 5f * Time.deltaTime;
        if (health <= 0) Die();

        healthText.text = "Health: " + health;
    }

    // Switch weapons
    void SwitchWeapon(int direction)
    {
        currentWeaponIndex += direction;
        if (currentWeaponIndex < 0) currentWeaponIndex = weapons.Length - 1;
        if (currentWeaponIndex >= weapons.Length) currentWeaponIndex = 0;

        foreach (var weapon in weapons) weapon.SetActive(false);
        weapons[currentWeaponIndex].SetActive(true);
    }

    // Shoot projectile
    void Shoot()
    {
        GameObject bullet = Instantiate(weapons[currentWeaponIndex].GetComponent<Weapon>().bulletPrefab, shootPoint.position, shootPoint.rotation);
        Rigidbody rb = bullet.GetComponent<Rigidbody>();
        rb.velocity = shootPoint.forward * bulletSpeed;
    }

    // Health system
    bool IsOutsideSafeZone()
    {
        return Vector3.Distance(transform.position, safeZone.position) > safeZone.localScale.x / 2;
    }

    void Die()
    {
        Debug.Log("You died!");
        Destroy(gameObject);
    }

    // Graphics Settings
    void OnQualityChanged(int index) => QualitySettings.SetQualityLevel(index);

    void OnResolutionChanged(int index)
    {
        Resolution res = Screen.resolutions[index];
        Screen.SetResolution(res.width, res.height, Screen.fullScreen);
    }

    int GetCurrentResolutionIndex()
    {
        Resolution currentRes = Screen.currentResolution;
        for (int i = 0; i < Screen.resolutions.Length; i++)
        {
            if (Screen.resolutions[i].width == currentRes.width && Screen.resolutions[i].height == currentRes.height)
                return i;
        }
        return 0;
    }

    // Ground collision detection
    void OnCollisionStay() { isGrounded = true; }
    void OnCollisionExit() { isGrounded = false; }
}
